## 11. Создание новой временной локальной таблицы из результирующего набора данных инструкции SELECT
```sql
SELECT 
    u.user_id,
    u.user_name,
    COUNT(s.sub_id) as subscription_count,
    SUM(s.price) as total_spent,
    CASE 
        WHEN COUNT(s.sub_id) = 1 THEN 'Level 1'
        WHEN COUNT(s.sub_id) BETWEEN 2 AND 4 THEN 'Level 2' 
        WHEN COUNT(s.sub_id) > 4 THEN 'Level 3'
        ELSE 'Level 0'
    END as user_level
INTO TEMP TABLE user_stats
FROM users u
LEFT JOIN subscriptions s ON u.user_id = s.user_id
GROUP BY u.user_id, u.user_name;
-- сначала выполнить создание, затем выполнить запрос
SELECT * FROM user_stats 
ORDER BY total_spent DESC, subscription_count DESC;

SELECT user_level, COUNT(*) as user_count
FROM user_stats
GROUP BY user_level
ORDER BY user_count DESC;

SELECT user_name, total_spent, subscription_count
FROM user_stats
WHERE total_spent > (SELECT AVG(total_spent) FROM user_stats)
ORDER BY total_spent DESC;
```
## 12. Инструкция SELECT, использующая вложенные коррелированные подзапросы в качестве производных таблиц в предложении FROM
```sql
WITH most_expensive AS (
    SELECT 
        'Most Expensive' AS criteria,
        u.user_name,
        s.service_name,
        sub.price
    FROM subscriptions sub
    JOIN users u ON sub.user_id = u.user_id
    JOIN services s ON sub.service_id = s.service_id
    WHERE sub.price = (SELECT MAX(price) FROM subscriptions)
),
longest_duration AS (
    SELECT 
        'Longest Duration' AS criteria,
        u.user_name,
        s.service_name,
        sub.price
    FROM subscriptions sub
    JOIN users u ON sub.user_id = u.user_id
    JOIN services s ON sub.service_id = s.service_id
    WHERE (sub.end_date - sub.start_date) = (
        SELECT MAX(end_date - start_date) 
        FROM subscriptions 
        WHERE start_date IS NOT NULL AND end_date IS NOT NULL
    )
)
SELECT * FROM most_expensive
UNION ALL
SELECT * FROM longest_duration;
```

## 22. Инструкция SELECT, использующая простое обобщенное табличное выражение
```sql
WITH UserSubscriptionStats AS (
    SELECT 
        user_id,
        COUNT(*) AS total_subscriptions,
        SUM(price) AS total_spent
    FROM subscriptions
    GROUP BY user_id
)
SELECT 
    u.user_name,
    u.email,
    uss.total_subscriptions,
    uss.total_spent
FROM UserSubscriptionStats uss
JOIN users u ON uss.user_id = u.user_id
WHERE uss.total_subscriptions >= 2
ORDER BY uss.total_spent DESC;
```
## 23. Инструкция SELECT, использующая рекурсивное обобщенное табличное выражение
```sql
WITH RECURSIVE referral_hierarchy AS (
    SELECT 
        u.user_id,
        u.user_name,
        ur.referred_id,
        1 as level
    FROM users u
    JOIN user_referrals ur ON u.user_id = ur.referrer_id
    UNION ALL
    SELECT 
        rh.user_id,
        rh.user_name,
        ur.referred_id,
        rh.level + 1
    FROM referral_hierarchy rh
    JOIN user_referrals ur ON rh.referred_id = ur.referrer_id
)
SELECT 
    user_id,
    user_name,
    COUNT(referred_id) as total_referrals_in_network
FROM referral_hierarchy
GROUP BY user_id, user_name
ORDER BY total_referrals_in_network DESC;
```
## 24. Оконные функции. Использование конструкций MIN/MAX/AVG OVER()
```sql
SELECT 
    user_id,
    service_id,
    price,
    AVG(price) OVER(PARTITION BY service_id) AS avg_service_price,
    MIN(price) OVER(PARTITION BY service_id) AS min_service_price,
    MAX(price) OVER(PARTITION BY service_id) AS max_service_price,
    RANK() OVER(PARTITION BY service_id ORDER BY price DESC) AS price_rank
FROM subscriptions
WHERE start_date IS NOT NULL;
```
## 25. Оконные функции для устранения дублей
```sql
-- Создали дубликаты промокодов

DELETE FROM promocodes
WHERE promocode_id IN (
    SELECT promocode_id
    FROM (
        SELECT promocode_id,
               ROW_NUMBER() OVER (
                   PARTITION BY service_id, promocode, sub_duration_days
                   ORDER BY promocode_id
               ) as rn
        FROM promocodes
    ) duplicates
    WHERE rn > 1
);
```

## Защита
Написать запрос, который показывает, сколько денег потеряли все компании из-за промокодов (поход в таблицу подписок и сложение всех цен подписок у которых написано промокод (по умолчанию промокод делает скидку на 1 месяц пользования)).
```sql
SELECT 
    s.service_id,
    s.service_name,
    COALESCE(SUM(sub.price), 0) AS total_loss
FROM services s
LEFT JOIN subscriptions sub ON s.service_id = sub.service_id 
    AND sub.sub_type = 'promocode'
GROUP BY s.service_id, s.service_name
ORDER BY total_loss DESC;
```

## Дополнительное задание 
```sql
-- Создание таблиц для версионности
CREATE TABLE Table1 (
    id INTEGER,
    var1 VARCHAR,
    valid_from_dttm DATE,
    valid_to_dttm DATE
);

CREATE TABLE Table2 (
    id INTEGER,
    var2 VARCHAR,
    valid_from_dttm DATE,
    valid_to_dttm DATE
);

-- Вставка тестовых данных
INSERT INTO Table1 VALUES 
(1, 'A', '2018-09-01', '2018-09-15'),
(1, 'B', '2018-09-16', '5999-12-31');

INSERT INTO Table2 VALUES 
(1, 'A', '2018-09-01', '2018-09-18'),
(1, 'B', '2018-09-19', '5999-12-31');

-- Версионное соединение
WITH all_dates AS (
    SELECT DISTINCT id, valid_from_dttm as date_point
    FROM Table1
    UNION
    SELECT DISTINCT id, valid_to_dttm as date_point
    FROM Table1
    UNION  
    SELECT DISTINCT id, valid_from_dttm as date_point
    FROM Table2
    UNION
    SELECT DISTINCT id, valid_to_dttm as date_point
    FROM Table2
),
date_ranges AS (
    SELECT 
        id,
        date_point as valid_from_dttm,
        LEAD(date_point) OVER (PARTITION BY id ORDER BY date_point) as valid_to_dttm
    FROM all_dates
    WHERE date_point < '5999-12-31'
)
SELECT 
    dr.id,
    t1.var1,
    t2.var2,
    dr.valid_from_dttm,
    COALESCE(dr.valid_to_dttm, '5999-12-31') as valid_to_dttm
FROM date_ranges dr
LEFT JOIN Table1 t1 ON dr.id = t1.id 
    AND dr.valid_from_dttm >= t1.valid_from_dttm 
    AND dr.valid_from_dttm < t1.valid_to_dttm
LEFT JOIN Table2 t2 ON dr.id = t2.id 
    AND dr.valid_from_dttm >= t2.valid_from_dttm 
    AND dr.valid_from_dttm < t2.valid_to_dttm
WHERE dr.valid_to_dttm IS NOT NULL
ORDER BY dr.valid_from_dttm;
```
